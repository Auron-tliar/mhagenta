 42    """Base class for agent module definitions. Contains templates for all basic functions that define module behaviour.
 43
 44    Extend one of the role-specific bases in ``mhagenta.bases`` to implement an agent module. Behaviour hooks
 45    (``step``, ``on_first``, ``on_last``, and message reactions) are synchronous and must return a ``State``.
 46    Keep them short and non-blocking: they run on the module's scheduling loop. ``on_init`` is a separate
 47    synchronous setup hook with no state return value.
 48
 49    Construction records a definition for later execution in a module subprocess. The runtime attaches the agent ID,
 50    state, and logger before invoking ``on_init``. They must not be used from a behaviour object's constructor.
 51    Hook return values are passed to the runtime, which processes queued outbox messages and termination requests.
 52

502    def save_state(self) -> None:
503        """Attempt to persist registered custom fields from ``State.dump()``.
504
505        Files are written beneath the configured save directory as ``<agent_id>.<module_id>.json`` for JSON or
506        ``<agent_id>.<module_id>.sav`` for dill. JSON uses the standard serializer without custom encoders; choose
507        dill for fields requiring Python object serialization and make their defining modules available on reload.
508
509        A new snapshot is serialized to ``<filename>.tmp`` before replacing the primary file. An existing primary
510        is first copied to ``<filename>.backup``. Save errors are logged as warnings and do not propagate to the
511        caller; a failed write attempt may leave a temporary file.
512
513        The runtime calls this after ``on_last`` and according to ``state_autosave_interval``: ``-1`` disables
514        autosaving, ``0`` saves each processed behaviour update, and a positive value schedules periodic saves
515        at that interval in seconds. ``on_init`` is not a state-returning behaviour update.
516        """
517        try:
518            path = Path(self._save_dir)
519            path.mkdir(exist_ok=True)
520            path /= f'{self._agent_id}.{self._module_id}.sav'
521            match self._save_format:
522                case 'json':
523                    path = path.with_suffix('.json')
524                    self._save_with_backup(path, json.dump, False)
525                case 'dill':
526                    self._save_with_backup(path, dill.dump)
527                case _:
528                    raise ValueError(f'Unsupported save format: {self._save_format}!')
529        except Exception as ex:
530            self.warning(f'Failed to save state! Reason: {type(ex)}({ex})! Resuming execution...')

532    def load_state(self, backup: bool = False) -> None:
533        """Read a saved snapshot and merge its custom fields into the current state.
534
535        Uses the configured format and the filenames described by ``save_state()``. Fields present in the snapshot
536        replace initial values; initial fields absent from the snapshot remain. Runtime fields and outbox contents
537        are not restored by normal snapshots.
538
539        When ``resume=True``, module construction first loads the primary file, then tries the backup if the primary
540        cannot be loaded. This method itself makes only the requested attempt. If both construction-time attempts
541        fail, the second failure propagates and the module cannot finish construction.
542        Deserialization and field-loading errors also propagate to the caller.
543
544        Args:
545            backup (bool, optional): Read ``<filename>.backup`` instead of the primary snapshot. Defaults to False.
546
547        Raises:
548            OSError: The requested file cannot be opened, including when it is missing.
549            ValueError: The configured save format is unsupported.
550        """
551        path = Path(self._save_dir) / f'{self._agent_id}.{self._module_id}.sav'
552        match self._save_format:
553            case 'json':
554                path = path.with_suffix('.json')
555                if backup:
556                    path = path.with_name(f'{path.name}.backup')
557                with open(path, 'r') as f:
558                    state = json.load(f)
559            case 'dill':
560                if backup:
561                    path = path.with_name(f'{path.name}.backup')
562                with open(path, 'rb') as f:
563                    state = dill.load(f)
564            case _:
565                raise ValueError(f'Unsupported save format: {self._save_format}!')
566
567        self._state.load(**state)
