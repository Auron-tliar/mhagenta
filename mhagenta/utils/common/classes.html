 178class TagCard[T](ABC):
 179    """
 180    Abstract base class for directory tag-based entries.
 181    """
 182    def __init__(self, tags: Iterable[T]) -> None:
 183        self.tags: set[T] = set(tags)
 184
 185    def match(self, query: Iterable[T]) -> bool:
 186        """Return whether every tag in ``query`` belongs to this card.
 187
 188        An empty query matches every card. Matching uses exact tag values.
 189        """
 190        for tag in query:
 191            if tag not in self.tags:
 192                return False
 193        return True

 198        pass

 201        return self.id == other

 257class BaseDirectory:
 258    """
 259    Base class for all directory types. Provides access by key and search by tag functionalities.
 260
 261    String keys select a card by its exact ID; integer keys select by insertion order, including negative indices.
 262    Membership checks such as ``module_id in directory`` test IDs. Lookups return cards, not ID strings.
 263
 264    Args:
 265        content (Iterable[TagCard[str]], optional): Initial cards. Supply a reusable collection such as a list or
 266            tuple, since construction traverses it to build both the ordered contents and the ID index.
 267    """
 268    def __init__(self, content: Iterable[TagCard[str]] | None = None) -> None:
 269        self._content = list(content) if content is not None else list()
 270        self._by_id = {card.id: card for card in content} if content is not None else dict()
 271
 272    def __getitem__(self, item: str | int) -> TagCard[str]:
 273        """Return a card by ID or insertion index.
 274
 275        Raises:
 276            KeyError: The string ID is not registered.
 277            IndexError: The integer index is outside the directory.
 278        """
 279        if isinstance(item, int):
 280            return self._content[item]
 281        else:
 282            return self._by_id[item]
 283
 284    def __contains__(self, item: str) -> bool:
 285        return item in self._by_id
 286
 287    def search(self, tags: Iterable[str]) -> list[TagCard[str]]:
 288        """Return cards containing all requested tags, in insertion order.
 289
 290        Args:
 291            tags (Iterable[str]): Exact tags to match. Supply a reusable collection such as a list or set because
 292                it is checked against every card. An empty collection selects all cards.
 293
 294        Returns:
 295            list[TagCard[str]]: Matching cards, or an empty list when no card matches.
 296        """
 297        output = list()
 298        for card in self._content:
 299            if card.match(tags):
 300                output.append(card)
 301        return output
 302
 303    def _add(self, card: TagCard[str]) -> None:
 304        self._content.append(card)
 305        self._by_id[card.id] = card
 306
 307    def __str__(self) -> str:
 308        return f'Directory(\n\t{'\n\t'.join([str(card) for card in self._content])})'
 309
 310    def __repr__(self) -> str:
 311        return str(self)

 803class Outbox(ABC):
 804    """
 805    Abstract base class for module outboxes.
 806
 807    Module-specific methods enqueue messages. After a behaviour hook returns its ``State``, the runtime sends the
 808    pending messages through the registered routes and clears them. Enqueuing a request does not wait for a reply;
 809    replies reach the corresponding message callback in a later invocation.
 810    """
 811    def __init__(self) -> None:
 812        self._msgs: dict[tuple[str, str, str | None], list[Any | dict[str, Any]]] = dict()
 813
 814        self._recipients = list()
 815        self._next_recipient = -1
 816        self._next_content = -1
 817
 818        self._term_request: str | None = None
 819
 820    def _add(self, recipient_id: str, performative: str, content: Any | dict[str, Any], extension: str = '') -> None:
 821        if (recipient_id, performative, extension) in self._msgs:
 822            self._msgs[recipient_id, performative, extension].append(content)
 823        else:
 824            self._msgs[recipient_id, performative, extension] = [content]
 825
 826    def __iter__(self) -> Self:
 827        self._recipients = list(self._msgs.keys())
 828        self._next_recipient = 0
 829        self._next_content = 0
 830        return self
 831
 832    def __next__(self) -> tuple[str, str, str, Any | dict[str, Any]]:
 833        if self._next_content >= len(self._msgs[self._recipients[self._next_recipient]]):
 834            self._next_recipient += 1
 835            self._next_content = 0
 836
 837        if self._next_recipient >= len(self._msgs):
 838            raise StopIteration
 839
 840        recipient_id, performative, extension = self._recipients[self._next_recipient]
 841        self._next_content += 1
 842        content = self._msgs[self._recipients[self._next_recipient]][self._next_content - 1]
 843        return recipient_id, performative, extension, content
 844
 845    def __bool__(self) -> bool:
 846        return bool(self._msgs)
 847
 848    def __str__(self) -> str:
 849        messages = list()
 850        for recipient, content_list in self._msgs.items():
 851            for content in content_list:
 852                messages.append(f'[TO: {recipient[0]}][{recipient[1]}{f"/{recipient[2]}" if recipient[2] else ""}]({content})')
 853        return f'Outbox({", ".join(messages)})'
 854
 855    def __repr__(self) -> str:
 856        return self.__str__()
 857
 858    def clear(self) -> None:
 859        """Discard queued messages and reset iteration, preserving any pending termination request."""
 860        self._msgs.clear()
 861        self._recipients = list()
 862        self._next_recipient = -1
 863        self._next_content = -1
 864
 865    def terminate_agent(self, reason: str = 'Manual termination') -> None:
 866        """
 867        Request an early stop of the whole agent through its root controller.
 868
 869        Call from the behaviour hook that detects completion and returns its state. The runtime processes this
 870        request after the hook returns, even when there are no outgoing messages. It reports the request to the root,
 871        which coordinates module shutdown and final hooks. This method does not stop the agent synchronously.
 872        Repeated calls before processing replace the pending reason. ``on_last`` already runs during shutdown.
 873
 874        Args:
 875            reason (str, optional): Reason attached to the request for logging. Defaults to ``'Manual termination'``.
 876        """
 877        self._term_request = reason
 878
 879    def pop_term_request(self) -> tuple[bool, str | None]:
 880        """Consume the pending termination request for the runtime.
 881
 882        Returns:
 883            tuple[bool, str | None]: Whether a request existed and its reason; ``(False, None)`` if none existed.
 884        """
 885        request = self._term_request
 886        self._term_request = None
 887        return request is not None, request
