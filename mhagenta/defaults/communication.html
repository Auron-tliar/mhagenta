 82        try:
 83            self.log(logging.DEBUG, f'Received message {msg.short_id} from {sender}.')
 84            update = self.on_message(self.state, sender, msg.body)

195    Extended perceptor base class for interacting with RabbitMQ-based environments.

197    Call ``observe()`` from a behaviour hook to request data, then handle the response in ``on_observation()``.
198    Requests and responses are dictionaries defined by the environment. To forward a result internally, construct
199    an ``Observation`` and enqueue it with ``state.outbox.send_observation()`` before returning the state.
200    Responses are addressed to the agent's observation route. Multiple perceptors subscribed to that route receive
201    the same responses; include and echo application correlation fields when a response must be matched to a request.
202    """

265        """
266        env_id = self.state.directory.external.environment.address['env_id'] if env_id is None else env_id
267        self.log(logging.DEBUG, f'Sending observation request to \"{env_id}\".')
268        self._connector.send(
269            recipient=env_id,
270            channel='',
271            msg=Message(
272                body=kwargs,
273                sender_id=self._agent_id,
