 18class MHAEnvBase:
 19    """
 20    Behaviour base-class for MHAEnvironment
 21
 22    Extend this class with synchronous observation and action handlers, then pass an instance to
 23    ``Orchestrator.add_environment()``. Each handler receives the runtime's current state dictionary; use that
 24    argument when constructing a replacement state. The runtime owns transport, timeout, and persistence.
 25    Keep handlers short and non-blocking because they run on the environment's event loop.
 26    """
 27    def __init__(self, init_state: dict[str, Any] | None) -> None:
 28        """Set the environment's initial state.
 29
 30        Args:
 31            init_state (dict[str, Any] | None): Initial state dictionary, kept by reference. This argument is required;
 32                pass ``None`` for a new empty dictionary. Environment state does not use the module ``State`` wrapper.
 33        """
 34        self.state = init_state if init_state is not None else {}
 35        self._log_func: Callable[[int, str], None] | None = None
 36
 37    def on_observe(self, state: dict[str, Any], sender_id: str, **kwargs) -> tuple[dict[str, Any], dict[str, Any]]:
 38        """
 39        Override to define what environment returns when observed by agents,
 40
 41        The default returns the unchanged state and an empty response. Every successful invocation returns a pair and
 42        sends the response dictionary, including an empty dictionary. Response keys become keyword arguments to
 43        subscribed RabbitMQ perceptors' ``on_observation`` hooks; no ``Observation`` wrapper is added automatically.
 44
 45        Args:
 46            state (dict[str, Any]): state of environment
 47            sender_id (str): sender agent id
 48            **kwargs: optional keyword parameters for observation action
 49
 50        Returns:
 51            tuple[dict[str, Any], dict[str, Any]]: tuple of modified state and keyword-based observation description
 52                response.
 53
 54        """
 55        return state, dict()
 56
 57    def on_action(self, state: dict[str, Any], sender_id: str, **kwargs) -> dict[str, Any] | tuple[dict[str, Any], dict[str, Any] | None]:
 58        """
 59        Override to define the effects of an action on the environment.
 60
 61        Returning only a state dictionary, or ``(state, None)``, updates the state without sending a response. Return
 62        ``(state, response)`` to send an action status; an empty response dictionary still produces a message.
 63        Response keys become keyword arguments to subscribed RabbitMQ actuators' ``on_status`` hooks.
 64        The default returns the unchanged state without a response.
 65
 66        Args:
 67            state (dict[str, Any]): state of environment
 68            sender_id (str): sender agent id
 69            **kwargs: keyword-based description of an action
 70
 71        Returns:
 72            dict[str, Any] | tuple[dict[str, Any], dict[str, Any] | None]: Updated state alone, or a pair of the updated
 73                state and an optional keyword-based action response.
 74
 75        """
 76        return state

 78    def log(self, level: int, message: str) -> None:
 79        """
 80        Log a message via the agent internal logging system
 81
 82        Args:
 83            level (int): log level
 84            message (str): log message
 85        """
 86        if self._log_func is None:
 87            raise TypeError('Log function has not been set properly during initialization')
 88        self._log_func(level, message)

 91class MHAEnvironment(MHABase, ABC):
 92    """
 93    Base class for MHAgentA environments
 94
 95    Transport implementations supply the async ``initialize``, ``on_start``, and ``on_stop`` methods and the
 96    synchronous ``send_response`` method. Application observation/action behaviour belongs in ``MHAEnvBase``.
 97    The environment launcher awaits ``initialize()`` and then ``start()``. A timeout or SIGINT/SIGTERM request
 98    runs the transport's stop hook and then saves the current state when a save directory is configured.
 99    """
100
101    def __init__(self,
102                 base: MHAEnvBase,
103                 env_id: str = "environment",
104                 exec_duration: float = 60.,
105                 start_time_reference: float | None = None,
106                 save_dir: os.PathLike | None = None,
107                 save_format: Literal['json', 'dill'] = 'json',
108                 log_id: str | None = None,
109                 log_tags: list[str] | None = None,
110                 log_level: int | str = logging.DEBUG,
111                 log_format: str = DEFAULT_LOG_FORMAT,
112                 tags: Iterable[str] | None = None
113                 ) -> None:
114        """Configure a transport-backed environment runtime.
115
116        Construction installs SIGINT and SIGTERM handlers and must take place in the main thread.
117
118        Args:
119            base (MHAEnvBase): Behaviour definition and initial state.
120            env_id (str, optional): Environment identifier used for routing and save filenames. Defaults to
121                ``'environment'``.
122            exec_duration (float, optional): Seconds to wait after ``start()`` before initiating shutdown.
123                Defaults to 60. The orchestrator may adjust this value when scheduling an environment.
124            start_time_reference (float, optional): Unix timestamp used as the origin for environment timestamps.
125                Defaults to construction time. This sets the clock reference; ``start()`` starts the timeout timer.
126            save_dir (os.PathLike, optional): Directory for the final state snapshot. Defaults to None, disabling
127                persistence. Path strings are also accepted by the underlying ``Path`` conversion.
128            save_format (Literal['json', 'dill'], optional): Snapshot serializer. Defaults to ``'json'``.
129            log_id (str, optional): Final logging tag. Defaults to the runtime class name.
130            log_tags (list[str], optional): Initial logging tags. Defaults to the environment ID.
131            log_level (int | str, optional): Logging threshold. Defaults to ``logging.DEBUG``.
132            log_format (str, optional): Logging format. Defaults to ``DEFAULT_LOG_FORMAT``.
133            tags (Iterable[str], optional): Environment tags for directory searches. Defaults to no tags.
134        """
135        super().__init__(
136            agent_id=env_id,
137            log_id=log_id,
138            log_tags=log_tags,
139            log_level=log_level,
140            log_format=log_format
141        )
142
143        self.id = env_id
144        self.tags = list(tags) if tags is not None else []
145        self._save_dir = Path(save_dir) if save_dir is not None else None
146        self._save_format = save_format
147        if start_time_reference is None:
148            start_time_reference = time.time()
149        self.time = AgentTime(
150            agent_start_ts=start_time_reference if start_time_reference is not None else time.time(),
151            exec_start_ts=start_time_reference if start_time_reference is not None else time.time()
152        )
153        self._exec_duration = exec_duration
154
155        self.base = base
156        self.base._log_func = self.log
157        self.state = base.state
158
159        self._main_task_group: asyncio.TaskGroup | None = None
160        self._stop_event = asyncio.Event()
161
162        signal.signal(signal.SIGINT, self.on_kill)
163        signal.signal(signal.SIGTERM, self.on_kill)
164
165    def set_start_time(self, start_time_reference: float | None) -> None:
166        self.time = AgentTime(
167            agent_start_ts=start_time_reference,
168            exec_start_ts=start_time_reference
169        )

179    async def start(self) -> None:
180        """Run the transport start hook and wait for timeout or a signal requesting shutdown.
181
182        Call ``initialize()`` first. This coroutine creates the task group used by transport implementations and
183        returns after its tasks finish, including the timeout/signal path through ``stop()``.
184        """
185        async with asyncio.TaskGroup() as tg:
186            self._main_task_group = tg
187            tg.create_task(self.on_start())
188            tg.create_task(self._timeout())

190    async def _timeout(self) -> None:
191        try:
192            await asyncio.wait_for(self._stop_event.wait(), timeout=self._exec_duration)
193        except TimeoutError:
194            pass
195        await self.stop()

201    async def stop(self) -> None:
202        """Await transport shutdown, then save the current state if persistence is configured."""
203        self.progress('Stopping environment...')
204        await self.on_stop()
205        if self._save_dir is not None:
206            self.log(logging.DEBUG, 'Saving environment state before exiting...')
207            self.save_state()

209    @abstractmethod
210    def send_response(self, recipient_id: str, channel: str, msg: Message, **kwargs) -> None:
211        """
212        Sends the response to an agent request.
213
214        Args:
215            recipient_id (str): recipient agent id
216            channel (str): communication channel (if used)
217            msg (Message): response message
218            **kwargs: additional keyword arguments for the response
219        """
220        pass
221
222    def _on_request(self, sender: str, channel: str, msg: Message) -> None:
223        match msg.performative:
224            case Performatives.OBSERVE:
225                self._on_observation_request(sender, channel, msg)
226            case Performatives.ACT:
227                self._on_action_request(sender, channel, msg)
228            case _:
229                self.warning(f'Received unknown message request: {msg.performative}! Ignoring...')
230
231    def _on_observation_request(self, sender: str, channel: str, msg: Message) -> None:
232        try:
233            self.state, response = self.base.on_observe(state=self.state, sender_id=sender, **msg.body)
234            msg = Message(
235                body=response,
236                sender_id=self.id,
237                recipient_id=sender,
238                ts=self.time.agent,
239                performative=Performatives.INFORM
240            )
241            self.send_response(recipient_id=sender, channel=f'observations', msg=msg)
242        except Exception as ex:
243            self.warning(f'Caught exception \"{ex}\" while processing observation request {msg.short_id} from {sender})!'
244                         f' Aborting processing and attempting to resume execution...')
245
246    def _on_action_request(self, sender: str, channel: str, msg: Message) -> None:
247        try:
248            result = self.base.on_action(state=self.state, sender_id=sender, **msg.body)
249            if isinstance(result, tuple):
250                self.state, response = result
251            else:
252                self.state = result
253                response = None
254            if response is not None:
255                msg = Message(
256                    body=response,
257                    sender_id=self.id,
258                    recipient_id=sender,
259                    ts=self.time.agent,
260                    performative=Performatives.INFORM
261                )
262                self.send_response(recipient_id=sender, channel=f'act_status', msg=msg)
263        except Exception as ex:
264            self.warning(f'Caught exception \"{ex}\" while processing action {msg.short_id} from {sender})!'
265                         f' Aborting processing and attempting to resume execution...')
266
267    def on_kill(self, signum: int | signal.Signals, frame: FrameType) -> None:
268        self._stop_event.set()

270    def save_state(self) -> None:
271        """Write the entire environment state to its configured directory.
272
273        The filename is ``<env_id>.json`` for JSON or ``<env_id>.sav`` for dill. JSON requires standard
274        JSON-serializable values. If no save directory is configured, nothing is written. Environment snapshots
275        are overwritten directly; this runtime does not implement the module backup/resume mechanism.
276        Filesystem and serialization errors propagate to the caller.
277
278        Raises:
279            ValueError: The configured save format is unsupported.
280        """
281        if self._save_dir is None:
282            return
283        path = self._save_dir
284        path.mkdir(exist_ok=True)
285        path /= f'{self.id}.sav'
286        match self._save_format:
287            case 'json':
288                path = path.with_suffix('.json')
289                with open(path, 'w') as f:
290                    json.dump(self.state, f)
291            case 'dill':
292                with open(path, 'wb') as f:
293                    dill.dump(self.state, f)
294            case _:
295                raise ValueError(f'Unsupported save format: {self._save_format}!')
296
297    @property
298    def _logger_extras(self) -> LoggerExtras | None:
299        return LoggerExtras(
300            agent_time=self.time.agent,
301            mod_time=self.time.module,
302            exec_time=str(self.time.exec) if self.time.exec is not None else '-',
303            tags=self.log_tag_str
304        )

Transport implementations supply the async initialize, on_start, and on_stop methods and the synchronous send_response method. Application observation/action behaviour belongs in MHAEnvBase. The environment launcher awaits initialize() and then start(). A timeout or SIGINT/SIGTERM request runs the transport's stop hook and then saves the current state when a save directory is configured.
