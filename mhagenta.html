 452    def _normalize_gpu_list(gpu_device_ids: str | int | Iterable[str | int] | Literal['all', 'any', 'none']) -> list[str] | Literal['all', 'any', 'none']:

 458            if gpu_device_ids in ('all', 'any', 'none'):
 459                return gpu_device_ids
 460            else:
 461                norm_list.append(str(gpu_device_ids))
 462        elif isinstance(gpu_device_ids, Iterable) and not isinstance(gpu_device_ids, str):
 463            for gpu_id in gpu_device_ids:
 464                if not isinstance(gpu_id, (int, str)) or isinstance(gpu_id, bool):
 465                    raise TypeError(f'Invalid GPU device ID type: {type(gpu_id)}')
 466                norm_list.append(str(gpu_id))
 467        return norm_list

1094    def _resolve_gpu_ids(gpu_device_ids: str | list[str]) -> list[DeviceRequest] | None:
1095        """Convert an optional GPU selection into Docker device requests."""
1096
1097        if len(gpu_device_ids) <= 0:
1098            return None
1099
1100        if isinstance(gpu_device_ids, str):
1101            match gpu_device_ids:
1102                case 'all':
1103                    return [
1104                        DeviceRequest(
1105                            driver="nvidia",
1106                            count=-1,
1107                            capabilities=[["gpu"]],
1108                        )
1109                    ]
1110                case 'any':
1111                    return [
1112                        DeviceRequest(
1113                            driver="nvidia",
1114                            count=1,
1115                            capabilities=[["gpu"]],

1117                    ]
1118                case 'none':
1119                    return None
1120                case _:
1121                    warnings.warn(f'Unrecognized GPU device IDs: {gpu_device_ids}')
1122                    return None
1123
1124        if len(set(gpu_device_ids)) != len(gpu_device_ids):
1125            raise ValueError("gpu_device_ids must not contain duplicates.")
1126
1127        return [
1128            DeviceRequest(
1129                driver="nvidia",
1130                device_ids=gpu_device_ids,
1131                capabilities=[["gpu"]],
1132            )
